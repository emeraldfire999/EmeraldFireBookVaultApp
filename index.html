"Found in this wing." : "Not in this wing."; } /* ⭐ Search all wings in the current library */ async function searchAllWings() { const title = document.getElementById("searchInput").value.trim(); const result = await findBookAcrossWings(title); document.getElementById("searchResult").textContent = result ? `Found in: ${result}` : "Not found in any wing."; } /* ⭐ Cross-wing fuzzy search (reads each wing’s library.json) */ async function findBookAcrossWings(titleToFind) { if (!currentLibraryHandle) return null; titleToFind = titleToFind.toLowerCase().trim(); for await (const entry of currentLibraryHandle.values()) { if (entry.kind !== "directory") continue; if (entry.name === "books") continue; try { const fileHandle = await entry.getFileHandle("library.json"); const file = await fileHandle.getFile(); const text = await file.text(); const data = JSON.parse(text || "[]"); // Handle both formats: array OR { books: [...] } const booksArray = Array.isArray(data) ? data : data.books; if (!Array.isArray(booksArray)) continue; for (const book of booksArray) { if (book.title && book.title.toLowerCase().includes(titleToFind)) { return entry.name; // FOUND IT (fuzzy match) } } } catch (err) { console.warn("Could not read", entry.name); } } return null; } /* ⭐ Library-wide fuzzy author search */ async function searchAuthorAcrossWings(authorToFind) { if (!currentLibraryHandle) return []; authorToFind = authorToFind.toLowerCase().trim(); const results = {}; // wingName → [book titles] for await (const entry of currentLibraryHandle.values()) { if (entry.kind !== "directory") continue; if (entry.name === "books") continue; try { const fileHandle = await entry.getFileHandle("library.json"); const file = await fileHandle.getFile(); const text = await file.text(); const data = JSON.parse(text || "[]"); const booksArray = Array.isArray(data) ? data : data.books; if (!Array.isArray(booksArray)) continue; for (const book of booksArray) { if (!book.author) continue; // Split semicolon-separated authors const authors = book.author.toLowerCase().split(";").map(a => a.trim()); // Fuzzy match ANY author in the list if (authors.some(a => a.includes(authorToFind))) { if (!results[entry.name]) results[entry.name] = []; results[entry.name].push(book.title); } } } catch (err) { console.warn("Could not read", entry.name); } } return results; // { wingName: [titles] } } function displayAuthorResults(results, authorName) { const output = document.getElementById("searchResult"); if (!results || Object.keys(results).length === 0) { output.textContent = `No books found for author: ${authorName}`; return; } let text = `${authorName} found in:\n\n`; for (const wing in results) { text += `• ${wing}\n`; for (const title of results[wing]) { text += ` - ${title}\n`; } text += "\n"; } output.textContent = text; } async function searchAuthor() { const author = document.getElementById("searchInput").value.trim(); const results = await searchAuthorAcrossWings(author); displayAuthorResults(results, author); }
